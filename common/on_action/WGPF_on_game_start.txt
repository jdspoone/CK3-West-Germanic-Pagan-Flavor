on_game_start = {
	on_actions = {
		update_west_germanic_traditions
	}
}

update_west_germanic_traditions = {
	effect = {
		every_culture_global = {

			# Replace Hirds with new West Germanic Warfare
			if = {
				limit = {
					has_cultural_pillar = heritage_west_germanic
					has_cultural_tradition = tradition_hird
				}
				remove_culture_tradition = tradition_hird
				add_culture_tradition = tradition_west_germanic_warfare
			}

			# Replace Witengamot with West Germanic Governance
			if = {
				limit = {
					has_cultural_pillar = heritage_west_germanic
					has_cultural_tradition = tradition_the_witenagemot
				}
				remove_culture_tradition = tradition_the_witenagemot
				add_culture_tradition = tradition_west_germanic_governance
			}

			# Replace City Keepers with Refined Poetry
			if = {
				limit = {
					has_cultural_pillar = heritage_west_germanic
					has_cultural_tradition = tradition_city_keepers
				}
				remove_culture_tradition = tradition_city_keepers
				add_culture_tradition = tradition_poetry

			}
		}
	}
}
